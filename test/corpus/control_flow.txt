export def Sum(values: list<number>): number
  var total = 0
  for value in values
    if value > 0
      total += value
    elseif value == 0
      continue
    else
      throw 'negative value'
    endif
  endfor
  return total
enddef

def Poll(): bool
  var attempts = 0
  while attempts < 3
    try
      attempts += 1
      return Fetch()
    catch /temporary/
      continue
    finally
      Log(attempts)
    endtry
  endwhile
  return false
enddef
